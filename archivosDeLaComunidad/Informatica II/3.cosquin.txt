El programa se ejecutara 5 veces, det la menor y la mayor distanciaal origen */

#include<iostream>
#include<cmath>

using namespace std;

struct cord
{
	float corx;
	float cory;
};

class punto
{
	public:
		punto();
		void valua(int cordx,int  cordy);
		struct cord muestra();
		float distOrigen();
		int testEje();      //ve si el punto esta sobre un eje
		int cuadrante();
	private:
		float x;
		float y;
};

punto::punto()
{
	x=y=0;
}

void punto::valua(int cordx, int cordy)
{	
	x=cordx;
	y=cordy;
}

struct cord punto::muestra()
{
	struct cord p;
		p.corx=x;
		p.cory=y;
	return p;
}

float punto::distOrigen()
{
	float b;
		b=(x*x)+(y*y);
	return sqrt(b);
}

int punto::testEje()
{
	if(x==0 || y==0)
	{
		return 1;
	}
	else
	{
		return 0;
	}
}

int punto::cuadrante()
{
	if(x>0 && y>0)
	{
		return 1;
	}
	else
	{
		if( x<0 && y>0 )
		{
			return 2;
		}
		else
		{
			if(x<0 && y<0 )
			{
				return 3;
			}
			else
			{
				if( x>0 && y<0)
				{
					return 4;
				}
				else
				{
					return 0;
				}
			}
		}
	}
}
