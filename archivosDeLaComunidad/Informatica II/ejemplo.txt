#include <stdio.h>
#include <stdlib.h>

struct nodo
{
       int dato;
       struct nodo *sig;
};

int menu(void);
void imprime(struct nodo *p);
int main()
{
    struct nodo *p, *primero, *actual, *previo;
    int opcion, info;
    
    primero=NULL;
    
    do
    {
        opcion=menu();
        switch (opcion)
        {
               case 1:
                    p=(struct nodo*)malloc(sizeof(*p));
                    if( p!=NULL )
                    {
                        printf("\n---Nuevo Nodo---\n");
                        printf("Ingrese un entero: ");
                        scanf("%d", &(*p).dato);
                        (*p).sig=NULL;
                        
                        previo=NULL;
                        actual=primero;
                        while ( actual !=NULL && (*p).dato>(*actual).dato )
                        {   
                            previo=actual;
                            actual=(*actual).sig; 
                        }

                        if( previo==NULL ) // el dato es el nuevo primero o el más pequeño de todos
                        {
                            (*p).sig=primero;
                            primero=p;
                        }
                        else 
                        { 
                              (*previo).sig=p;
                              (*p).sig=actual;
                        }
                    }
                    else
                    {
                        printf("No hay memoria\n");
                        return 1;
                    }
                    break;
               case 2:
                    if(  primero!=NULL )  // averigua si hay nodos en la estructura
                    { 
                         printf("Ingresar el nodo a borrar\n");           
                         scanf ( "%d",&info); 
                         if( info==(*primero).dato)
                         { 
                             p=primero;
                             primero=(*p).sig;
                             free(p);
                         }
                         else 
                         { 
                              previo=primero;
                              actual=(*primero).sig;           // se posicionan los punteros para la búsqueda    
                         }
                         while ( actual!=NULL &&  (*actual).dato!=info)
                         {   
                              previo=actual;
                              actual=(*actual).sig; }

                              if( actual!=NULL ) 
                              { 
                                  p=actual;
                                  (*previo).sig= (*actual).sig;
                                  free(p);
                              }
                              else     
                              {
                                   printf ("no lo encontró\n");
                              }                
                         } 
                     else 
                     { 
                          printf ( " lista vacia");
                     }
                    break;
               case 3:
                    if(primero!=NULL)
                    {
                          p=primero;
                          imprime(p);
                    }
                    else
                    {
                        printf("La lista esta vacia\n");
                    }
                    break;
               case 4:
                    system("CLS");
                      printf("FIN DEL PROGRAMA\n\n");
                    break;
               default:
                    system("CLS");
                      printf("¡ERROR! INgrese una opcion valida\n");
                    break;
        }
    }while(opcion!=4);
  
  
  system("PAUSE");	
  return 0;
}
