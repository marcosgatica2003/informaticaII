{
	if((*enredadera)==NULL)
		{ 	*enredadera=new(struct arbol);

			(*enredadera)->izq=NULL;
			(*enredadera)->der=NULL;
		}
	else 
		{ 	if(((*enredadera)->dato)<a)
				{ carga(&(*enredadera)->izq,a);}
			else

		}
}
