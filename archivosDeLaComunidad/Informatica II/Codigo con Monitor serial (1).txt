#if defined(ARDUINO) && ARDUINO >= 100
  #include "Arduino.h"
  #include "SoftwareSerial.h"
  SoftwareSerial port(12,13);
#else 
  #include "WProgram.h"
  #include "NewSoftSerial.h"
  NewSoftSerial port(12,13);
#endif

#include "EasyVR.h"
EasyVR easyvr(port);

enum Groups
{
  GROUP_1  = 1,
};

enum Group1 
{
  G1_HOLA = 0,
  G1_UNO = 1,
  G1_DOS = 2,
  G1_TRES = 3,
  G1_CUATRO = 4,
  G1_CINCO = 5,
  G1_SEIS = 6,
  G1_CHAU = 7,
  G1_APLAUSO = 8,
};

EasyVRBridge bridge;

int8_t group, idx;
int    leds[5]={45, 47, 49, 51, 53};
int    rele=40;
int    estado=0;
int    i, j;

void prende(int);
void apaga(int);
void parpadea();
void action();

void setup()
{  
  // ---- Setup de los pin de los leds ---- //
  for(i=0 ;i<5 ;i++)
  {
      pinMode(leds[i], OUTPUT);
  }
 
  // ---- Setup del pin del rele ---- //
  pinMode(rele, OUTPUT);
  
  // ---- Setup del Easy VR ---- //
  if (bridge.check())
  {
    cli();
    bridge.loop(0, 1, 12, 13);
  }
  
  Serial.begin(9600);
  port.begin(9600);
    
  while(!easyvr.detect())
  {
    Serial.println("EasyVR not detected!");
    delay(100);
  }
  
  easyvr.setPinOutput(EasyVR::IO1, LOW);
  Serial.println("EasyVR detected!");
  prende(5);
  delay(200);
  parpadea();
  easyvr.setTimeout(5);
  easyvr.setLanguage(0);

  group = EasyVR::TRIGGER; 
}

void loop()
{
  group=GROUP_1;
  easyvr.setPinOutput(EasyVR::IO1, HIGH); // El Easy VR empieza a "escuchar"
  Serial.print("Say a command in Group ");
  Serial.println(group);
  easyvr.recognizeCommand(group);

  do
  {
  }
  while (!easyvr.hasFinished());
  
  easyvr.setPinOutput(EasyVR::IO1, LOW); // LED off

  idx = easyvr.getWord();
  if (idx >= 0)
  {
    return;
  }
  idx = easyvr.getCommand();
  if (idx >= 0)
  {
    // print debug message
    uint8_t train = 0;
    char name[32];
    Serial.print("Command: ");
    Serial.print(idx);
    if (easyvr.dumpCommand(group, idx, name, train))
    {
      Serial.print(" = ");
      Serial.println(name);
    }
    else
      Serial.println();
    easyvr.playSound(0, EasyVR::VOL_FULL);
    
    action();
  }
  else // errors or timeout
  {
    if (easyvr.isTimeout())
      Serial.println("Timed out, try again...");
      prende(1);
      apaga(1);
    int16_t err = easyvr.getError();
    if (err >= 0)
    {
      Serial.print("Error ");
      Serial.println(err, HEX);
      prende(2);
      apaga(2);
    }
  }
}

void prende(int num)
{
    for(i=0 ;i<num ;i++)
    {
        digitalWrite(leds[i], HIGH);
        delay(80);
    }
}

void apaga(int num)
{
    for(i=0 ;i<num;i++)
    {
        digitalWrite(leds[i], LOW);
    }
}

void parpadea()
{
    for(j=0;j<2;j++)
    {
      for(i=0 ;i<5 ;i++)
      {
          digitalWrite(leds[i], HIGH);
      }
      delay(150);
      
      for(i=0 ;i<5 ;i++)
      {
          digitalWrite(leds[i], LOW);
      }
      delay(150);
    }
}

void ReleParpadea(int num)
{
   if(estado==0)
   {
       for(i=0 ;i<num ;i++)
       {
             digitalWrite(rele, HIGH);
             delay(500);
             digitalWrite(rele, LOW);
             delay(500);
        }
    }
    else
    {
        for(i=0 ;i<num ;i++)
       {
             digitalWrite(rele, LOW);
             delay(500);
             digitalWrite(rele, HIGH);
             delay(500);
        }
    }
}
  

void action()
{
    switch (group)
    {
       case GROUP_1:
           parpadea();
            
           switch (idx)
           {
              case G1_HOLA:
                  digitalWrite(rele, HIGH);
                  estado=1;
                break;
              case G1_UNO:
                  ReleParpadea(1);
                break;
              case G1_DOS:
                  ReleParpadea(2);
                break;
              case G1_TRES:
                  ReleParpadea(3);
                break;
              case G1_CUATRO:
                  ReleParpadea(4);
                break;
              case G1_CINCO:
                  ReleParpadea(5);
                break;
              case G1_SEIS:
                  ReleParpadea(6);
                break;
              case G1_CHAU:
                  digitalWrite(rele, LOW);
                  estado=0;
                break;
              case G1_APLAUSO:
                  if(estado==0)
                  {
                      digitalWrite(rele, HIGH);
                      estado=1;
                  }
                  else
                  {
                      digitalWrite(rele, LOW);
                      estado=0;
                  }
                break;
              }
        break;
    }
}
