#include <stdio.h>
#include <stdlib.h>
#define tam 3

struct nodo1
{
	float concentracion;
	struct nodo1 *sig;
};

struct nodo2
{
	float nivel;
	struct nodo2 *sig;
};

struct nodo3
{
	float concentracion;
	float nivel;
	struct nodo3 *sig;
};

int main()
{
	struct nodo1 *p1, *base1;
	struct nodo2 *p2, *base2;
	struct nodo3 *p3, *base3;
	int  i;
	
	base1=NULL;
	base2=NULL;
	base3=NULL;
	
	for(i=0 ;i<tam ;i++)
	{
		printf("\n----- Lectura Nro: %d -----\n", i+1);
		p1=(struct nodo1*)malloc(sizeof(*p1));
		if(p1==NULL)
		{
			printf("No hay memoria\n");
			return 1;
		}
		else
		{
			printf("Ingrese la concentrecion de gas: ");
			scanf("%f", &(*p1).concentracion);
			(*p1).sig=base1;
			base1=p1;
		}
		p2=(struct nodo2*)malloc(sizeof(*p2));
		if(p2==NULL)
		{
			printf("No hay memoria\n");
			return 1;
		}
		else
		{
			printf("Ingrese el nivel de gas en el cartucho: ");
			scanf("%f", &(*p2).nivel);
			(*p2).sig=base2;
			base2=p2;
		}	
	}
	
	while( (p1!=NULL) && (p2!=NULL) )
	{
		p3=(struct nodo3*)malloc(sizeof(*p3));
		if(p3==NULL)
		{
			printf("No hay memoria\n");
			return 1;
		}
		else
		{
			(*p3).concentracion=(*p1).concentracion;
			(*p3).nivel=(*p2).nivel;
			
			(*p3).sig=base3;
			base3=p3;
			
			p1=(*p1).sig;
			p2=(*p2).sig;
		}
	}
	
	p1=base1;
	p2=base2;
	system("CLS");

	printf("Lista de lecturas de concentracion de gas:\n");
	while(p1!=NULL)
	{
		printf("%.2f \t", (*p1).concentracion);
		p1=(*p1).sig;	
	}
	
	printf("\n\nLista de lecturas de nivel de liquido:\n");
	while(p2!=NULL)
	{
		printf("%.2f \t", (*p2).nivel);
		p2=(*p2).sig;
	}
	
	printf("\n\nLista de lecturas juntas: \n");
	while(p3!=NULL)
	{
		printf("Concentracion: %.2f     Nivel:%.2f\n",(*p3).concentracion, (*p3).nivel);
		p3=(*p3).sig;
	}
	
	
	system("PAUSE");
	return 0;
}
